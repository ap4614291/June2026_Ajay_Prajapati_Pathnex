### Ansible - Install Java & Set JAVA_HOME

---
- name: Install Java & Set JAVA_HOME
  hosts: test
  become: true

  tasks:
    - name: Install Java
      package:
        name: java-21-openjdk
        state: present
    
    - name: Set JAVA_HOME
      lineinfile:
        path: /etc/profile
        line: 'export JAVA_HOME=/usr/lib/jvm/java21-openjdk'

### Jenkins Pipeline - Slack Notification

pipeline {

  agent {
    label 'worker-1'
  }

  environment{
    INSTITUTE_NAME = "HCL"
    SLACK_CHANNEL = "#devops"
  }

  stages {
    stage('Checkout'){
        steps {
            git branch: "dev" , url: 'https://github.com/Pathnex/sample-java-app.git' 
        }
      }
    stage('Build'){
      steps {
        sh 'mvn clean package'
      }
    }
  }

  post {
    success {
      slackSend(channel: "$SLACK_CHANNEL", message: "SUCCESS: $INSTITUTE_NAME Build #${env.BUILD_NUMBER}")
    }
    failure{
      slackSend(channel: "$SLACK_CHANNEL", message: "FAILURE: $INSTITUTE_NAME Build #${env.BUILD_NUMBER}")
    }
  }
}


### GitLab CI - Slack Notification

  - build

  INSTITUTE_NAME: "HCL"
  SLACK_WEBHOOK: "https://hooks.slack.com/services/XXXXX/YYYYY/ZZZZZ"

  stage: build
