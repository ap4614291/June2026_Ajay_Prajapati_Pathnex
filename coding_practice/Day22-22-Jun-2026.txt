### Anisble - User Handler on Config Change

---
- name: Configure Nginx for Pathnex
  hosts: webservers
  become: true

  tasks:
    - name: Copy Nginx Config
      copy:
        dest: /etc/nginx/nginx.conf
        content: |
          user nginx;
          worker_processes auto;
      notify: Restart Nginx
  handlers:
    - name: Restart Nginx
      service:
        name: nginx
        state: restarted


### Terraform - Security Group

provider "aws" {
  region = "us-east-1"
}

resource "aws_security_group" "hcl_sg" {

  name = "hcl_sg"
  vpc_id = aws_vpc.hcl_vpc.id

  ingress{
    from_port = 22
    to_port = 22
    protocol = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  egress{
    from_port = 0
    to_port = 0
    protocol = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }

  tags = {
    Name = "hcl_sg"
  }
}

### Jenkins Pipeline — Trigger on Git Tag

pipeline {
  
  agent {
    label 'worker-1'
  }

  environment {
    INSTITUTE_NAME = "Pathnex"
    TEAM = "DevOps"
    ENV = "prod"
  }

  triggers {
    pollSCM('H/15 * * * *')
  }

  stages{
    stage('Checkout'){
      steps{
        git branch: 'main',
            url: 'https://github.com/Pathnex/sample-java-app.git'
      }
    stage('Build'){
      when {
        buildingTag()
      }
      steps{
        sh 'echo "Building $INSTITUTE_NAME app for $TEAM in $ENV environment"'
      }
    }
    }
  }
}
