### Ansible - Create Mutliptle Users in Loop
---
- name: Create multiple Users
  hosts: all
  become: true

  vars:
    users:
      - dev1
      - dev2
      - dev3
  
  tasks:
    - name: Create Users 
      user:
        name: "{{ item }}"
        state: present
      loop: "{{ users }}"

### Kubernetes - Namespace + Deployment

#ns.yml
apiVersion: v1
kind: Namespace
metadata:
  name: hcl-ns

#dep.yml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: hcl-dep-app
  namespace: hcl-ns
spec:
  replicas: 2
  selector:
    matchLabels:
      app: hcl-app
  template:
    metadata:
      labels:
        app: hcl-app
    spec:
      containers:
        - name: hcl-container
          image: nginx:latest
          ports:
            - containerPort: 80
          resources:
            requests:
              cpu: "100m"
              memory: "250Mi"
            limits:
              cpu: "200m"
              memory: "500Mi"

### Jenkins Pipeline - Simulated Deployment

pipeline{

  agent {
    label "worker-1"
  }

  environment{
    INSTITUTE_NAME = "HCL"
  }

  parameters{
    choice(name: 'ENVIRONMENT', choices: ['dev','staging','prod'], description: 'Deployment Environment')
  }

  stages{
    stage('Deploy'){
      steps{
        sh 'echo "Deploying $INSTITUTE_NAME application to $ENVIRONMENT"'
      }
    }
  }
}
