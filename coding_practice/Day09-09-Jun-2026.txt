## Ansible - Use Jinja2 Template

---
- name: Deploye template for Pathnex
	hosts: all
	become: true

	tasks:
		- name: Copy template file
			template:
				src: pathnex.conf.j2
				dest: /etc/pathnex.conf



## Terraform - Internet Gateway

provider "aws" {
		region = "us-east-1"
}

resource "aws_vpc" "PathnexVPC"{

	cidr_block = "10.0.0.0/16"	

	tags = {
		Name = "PathnexVPC"
	}
}

resource "aws_internet_gateway" "PathnexIGW" {

	vpc_id = aws_vpc.PathnexVPC.id

	tags = {
		Name = "PathnexIGW"
	}
}

## Jenkins Pipeline - Archive Artifacts

pipeline{
	
	agent any

	tools {
		maven "Maven-3.8.1-jdk-17"
	}

	stages{
		stage('Checkout'){
			steps{
				git branch: "main", url: 'https://github.com/Pathnex/sample-java-app.git'
			}
		}
		stage('Build'){
			steps{
				sh 'mvn clean package'
			}
		}
		stage('Archive'){
			steps{
				archiveArtifacts artifacts: 'target/*.jar', fingerprint: true
			}
		}
	}
}



## GitLab CI - Artifacts

stages:
	- build

build:
	stage: build
	image: maven:3.8.1-jdk-17
	script:
		- git clone https://github.com/Pathnex/sample-java-app.git 
		- cd sample-java-app
		- mvn clean package

	artifacts:
		paths:
			- target/*.jar
		expire_in: 1 week
